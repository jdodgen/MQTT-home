import asyncio
import socket
import struct

# Official UPnP Multicast configuration parameters
MULTICAST_GROUP = "239.255.255.250"
MULTICAST_PORT = 1900

def get_local_ip() -> str:
    """Helper function to discover the computer's primary local IP network address."""
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        # Pings a public address (doesn't send actual data) to see which local interface is active
        s.connect(("8.8.8.8", 80))
        local_ip = s.getsockname()[0]
    except Exception:
        local_ip = "127.0.0.1"
    finally:
        s.close()
    return local_ip

async def start_ssdp_listener():
    """
    Listens for Alexa's UDP network discovery packets on Port 1900.
    Replies with the location of your Port 80 /description.xml sub-route.
    """
    local_ip = get_local_ip()
    
    # 1. Create a UDP socket configured for network multicast reuse
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    
    # Bind directly to all incoming interfaces on the standard SSDP discovery port
    sock.bind(("", MULTICAST_PORT))
    
    # Join the official UPnP multicast group
    mreq = struct.pack("4sl", socket.inet_aton(MULTICAST_GROUP), socket.INADDR_ANY)
    sock.setsockopt(socket.IPPROTO_IP, socket.IP_ADD_MEMBERSHIP, mreq)
    sock.setblocking(False)
    
    print(f"SSDP Listener active. Advertising Hue Bridge route via http://{local_ip}:80/description.xml")

    # 2. Build the exact HTTP over UDP response headers Alexa expects
    ssdp_reply = (
        "HTTP/1.1 200 OK\r\n"
        "CACHE-CONTROL: max-age=120\r\n"
        "EXT:\r\n"
        f"LOCATION: http://{local_ip}:80/description.xml\r\n"
        "SERVER: Linux/3.14.0 UPnP/1.0 IpBridge/1.24.0\r\n"
        "ST: urn:schemas-upnp-org:device:basic:1\r\n"
        "USN: uuid:2f402f80-da50-11e1-9b23-00178825102a::urn:schemas-upnp-org:device:basic:1\r\n"
        "\r\n"
    ).encode('utf-8')

    loop = asyncio.get_running_loop()

    # 3. Process incoming discovery messages without blocking the main event loop
    while True:
        try:
            data, addr = await loop.sock_recvfrom(sock, 1024)
            message = data.decode('utf-8', errors='ignore')
            
            # Alexa triggers a scan by broadcasting an M-SEARCH query packet
            if "M-SEARCH" in message and ("ssdp:all" in message or "device:basic" in message):
                # Send the response back directly to the Echo device that requested it
                await loop.sock_sendto(sock, ssdp_reply, addr)
                print(f"Responded to Alexa discovery scan from Echo device at {addr[0]}")
                
        except asyncio.CancelledError:
            break
        except Exception as e:
            print(f"SSDP Error: {e}")
            await asyncio.sleep(1)
            
    sock.close()

Since your web server runs inside an asyncio event loop via aiohttp, you shouldn't use publish.single() directly because it is synchronous and would block your web server. Instead, we can wrap it using asyncio.to_thread().
Here is how to complete the control circuit.
## Step 1: Add the Asynchronous MQTT Sender
Add this helper function to your main_http.py script to handle non-blocking MQTT publishing:

import asyncioimport paho.mqtt.publish as publishimport config # Using your existing configuration structure
async def async_mqtt_publish(topic: str, payload: str):
    """
    Executes paho-mqtt in a background thread so it doesn't block 
    the main aiohttp web server loop.
    """
    # Replace with your actual broker IP/Port setup
    MQTT_BROKER = "127.0.0.1" 
    MQTT_PORT = 1883
    
    # Runs the synchronous publish function inside a worker thread
    await asyncio.to_thread(
        publish.single,
        topic=topic,
        payload=payload,
        hostname=MQTT_BROKER,
        port=MQTT_PORT
    )
    print(f"MQTT Broadcast sent to {topic} -> {payload}")

async def handle_set_light_state(request):
    """
    Translates Alexa's numeric command ID, updates local state, 
    and publishes the action out over MQTT.
    """
    device_id = request.match_info.get('id') 
    body = await request.json()
    target_state = body.get("on") # True or False

    internal_keys = list(MY_INTERNAL_DEVICES.keys())
    
    try:
        array_index = int(device_id) - 1
        matched_device_key = internal_keys[array_index]
        
        # 1. Update your local state tracking engine
        new_state_string = "ON" if target_state else "OFF"
        MY_INTERNAL_DEVICES[matched_device_key]["current_state"] = new_state_string
        
        # 2. CONSTRUCT YOUR TARGET MQTT TOPIC AND PAYLOAD
        # Example 1: Standard Zigbee2MQTT device topic layout
        # target_topic = f"zigbee2mqtt/{matched_device_key}/set"
        # mqtt_payload = json.dumps({"state": new_state_string})
        
        # Example 2: Simple state string topic layout
        target_topic = f"home/actuators/{matched_device_key}/state"
        mqtt_payload = new_state_string
        
        # 3. CHOOSE NON-BLOCKING EXECUTION
        # Fire-and-forget background execution (Fastest UI response to Alexa)
        asyncio.create_task(async_mqtt_publish(target_topic, mqtt_payload))

## How the full data loop operates now

   1. User Voice Input: You say, "Alexa, turn on the Living Room Light".
   2. Alexa Request: Your Echo speaker sends a PUT message straight to http://<your_ip>:80/api/api_token_12345/lights/1/state with {"on": true}.
   3. Sub-Route Conversion: aiohttp catches it, converts ID "1" back into "living_room_lamp", and spins up an async worker.
   4. MQTT Dispatch: The script constructs the target topic and hands off the message payload smoothly without freezing your web panel.
   5. Hardware Updates: Zigbee2MQTT (or your target subscriber) intercepts the topic and immediately toggles the physical device.
